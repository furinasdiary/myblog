#include<bits/stdc++.h>
using namespace std;
#define int long long

bool isprime[100000000];
int prime[100000000];

signed main(){
    ios::sync_with_stdio(0);
    cin.tie(0);    
    int n;
    cin >> n;
    for(int i=2;i<=n;i++){
        isprime[i]=1;
    }
    int m=0;
    for(int i=2;i<=n;i++){
        if(isprime[i]) prime[++m]=i;
        for(int j=1;j<=m&&prime[j]*i<=n;j++){
            isprime[prime[j]*i]=0;
            if(i%prime[j]==0)break;
        }
    }
    for (int i=1;i<=m;i++){
        cout<<prime[i]<<" ";
    }
    return 0;
}
