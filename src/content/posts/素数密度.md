#include<bits/stdc++.h>
using namespace std;
#define int long long
#define all(x) x.begin(),x.end()
#define YES cout<<"YES\n"
#define NO cout<<"NO\n"
#define cfs cout<<fixed<<setprecision
typedef pair<int,int> PII;
const int N=2e5+10;
const int INF=1e18;

int prime[N];
int isprime[N];
bool vis[10000000];

void solve(){
    int l,r;
    cin>>l>>r;
    for(int i=2;i<N;i++){
        isprime[i]=1;
    }
    int m=0;
    for(int i=2;i<N;i++){
        if(isprime[i]) prime[++m]=i;
        for(int j=1;j<=m&&i*prime[j]<N;j++){
            isprime[i*prime[j]]=0;
            if(i%prime[j]==0) break;
        }
    }
    for(int i=1;i<=m;i++){
        int p=prime[i];
        int start=max(p*2,(int)ceil(l/p)*p);
        for(int j=start;j<=r;j+=p){
            if(j-l>=0) vis[j-l]=1;
        }
    }
    int ans=0;
    vis[1]=1;
    for(int i=0;i<=r-l;i++){
        if(!vis[i]) ans++;
    }
    cout<<ans;
    
}
signed main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    int T=1;
    //cin>>T;
    while(T--) solve();
    return 0;
}
